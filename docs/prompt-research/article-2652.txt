          video: this.uploadedVideo

        success: (res) => {

          }
          $('.submit-btn').prop('disabled', false).text('鍙戣〃璇勮');

    // 鍔犺浇璇勮鍒楄〃
    loadComments(append = false) {
      if (!append) {
        $('.comment-loading').show();
        $('.comment-items').html('');
      }

      $.ajax({
        url: 'common/comment-api.php?action=get_comments&info_id=' + this.infoId + '&page=' + this.currentPage + '&pagesize=' + this.pageSize,
        type: 'GET',
        dataType: 'json',
        success: (res) => {
          $('.comment-loading').hide();
          
          if (res.code === 200) {
            // 鏄剧ず鎵€鏈夎瘎璁烘€绘暟锛堝寘鎷洖澶嶏級
            const totalCount = res.total_all || res.total;
            $('.comment-count').text(totalCount);
            
            if (res.data.length === 0 && !append) {
              $('.comment-empty').show();
              $('.comment-items').hide();
              $('.comment-loadmore').hide();
            } else {
              $('.comment-empty').hide();
              $('.comment-items').show();
              
              const html = res.data.map(comment => this.renderComment(comment)).join('');
              if (append) {
                $('.comment-items').append(html);
              } else {
                $('.comment-items').html(html);
              }

              // 鏄剧ず/闅愯棌鍔犺浇鏇村鎸夐挳
              if (res.data.length < this.pageSize) {
                $('.comment-loadmore').hide();
              } else {
                $('.comment-loadmore').show();
              }
            }
          }
        },
        error: () => {
          $('.comment-loading').hide();
          Toast.show('鍔犺浇璇勮澶辫触', 'error');
        }
      });
    },

    // 鏍煎紡鍖栨椂闂?- 灏忕孩涔﹂鏍?    formatTime(time) {
      const date = new Date(time);
      const now = new Date();
      const diff = Math.floor((now - date) / 1000); // 绉掑樊
      
      if (diff < 60) {
        return '鍒氬垰';
      } else if (diff < 3600) {
        const minutes = Math.floor(diff / 60);
        return `${minutes}鍒嗛挓鍓峘;
      } else if (diff < 86400) {
        const hours = Math.floor(diff / 3600);
        return `${hours}灏忔椂鍓峘;
      } else if (diff < 604800) {
        const days = Math.floor(diff / 86400);
        return `${days}澶╁墠`;
      } else {
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        return `${month}-${day}`;
      }
    },

        "'": '&#039;'
      };
      return text.replace(/[&<>"']/g, m => map[m]);
